POST /upload
  Content-Type : multipart/form-data
  Body field   : file  (binary, .csv or .xlsx)
  Success 200  : { session_id: string, metadata: DatasetMetadata }
  Error   422  : DataIngestionError (bad file / unsupported type)
  Error   500  : Unexpected server error

POST /analyze
  Content-Type : application/json
  Body schema  :
    {
      "session_id"     : "<uuid>",          // required — from /upload
      "model_provider" : "ollama",          // "ollama" | "openai" | "anthropic"
      "model_name"     : "gemma:2b",        // optional; provider default if omitted
      "use_csvl"       : true,              // closed-loop self-validation toggle
      "num_insights"   : 10,               // 1–30
      "openai_key"     : null,             // optional runtime key
      "anthropic_key"  : null              // optional runtime key
    }
  Success 200  : AnalyzeResponse (metrics + validation_results + ground_truth)
  Error   404  : session not found
  Error   500  : pipeline failure (LLM error, etc.)

GET /results?session_id=<uuid>
  Success 200  : { status: "complete" | "uploaded" | "error", ...AnalyzeResponse }
  Error   404  : session not found
